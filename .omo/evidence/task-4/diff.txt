   130	function CompletenessBar({ value }: { value: number | null }) {
   131	  if (value === null)
   132	    return <span className="text-slate-500 text-xs">no snapshot</span>;
   133	  const pct = Math.round(value * 100);
   134	  const color =
   135	    pct >= 60 ? 'bg-green-500' : pct >= 30 ? 'bg-yellow-500' : 'bg-red-500';
   136	  return (
   137	    <div className="flex items-center gap-2">
   138	      <div className="w-16 h-1.5 bg-slate-800 rounded overflow-hidden">
   139	        <div className={`h-full ${color}`} style={{ width: `${pct}%` }} />
   140	      </div>
   141	      <span className="text-xs text-slate-400">{pct}%</span>
   142	    </div>
   143	  );
   144	}
